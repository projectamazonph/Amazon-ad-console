// ============================================================================
// AMAZON PPC CONSOLE SIMULATOR — DATA SCHEMA
// ============================================================================
// Design notes:
// - Mirrors Amazon Ads' actual entity hierarchy: Portfolio > Campaign > AdGroup
//   > Targeting/Ads, so the mock data engine and UI share one ground truth.
// - CampaignType (SP/SB/SD) + CampaignSubtype together gate which child tables
//   and which wizard steps are valid — this is the branching logic discussed
//   earlier (SB format fork, SD targeting-tactic fork, SP targeting-type fork).
// - PerformanceMetric is a single polymorphic daily rollup table reused across
//   every entity level (campaign/ad group/keyword/product/search term) so the
//   mock data engine and reporting dashboards only need one writer/reader.
// - ScenarioTask + ScenarioCheck are the training layer: they validate against
//   the *state* of an account, not screenshots, so scoring is deterministic.
// ============================================================================

generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

// ----------------------------------------------------------------------------
// ENUMS
// ----------------------------------------------------------------------------

enum CampaignType {
  SP // Sponsored Products
  SB // Sponsored Brands
  SD // Sponsored Display
}

// Only meaningful combos per CampaignType are enforced at the application
// layer (see CAMPAIGN_SUBTYPE_MATRIX below in comments), not via DB constraint,
// since Prisma enums can't be conditionally scoped to another column.
enum CampaignSubtype {
  // SP — targeting type chosen at ad-group level, but stored on campaign too
  // for filtering/reporting convenience. "MIXED" allowed only for SP since
  // ad groups within one SP campaign could theoretically differ (rare, but
  // the real console allows it — good edge case for scenario mode).
  SP_AUTO
  SP_MANUAL_KEYWORD
  SP_MANUAL_PRODUCT
  SP_MIXED

  // SB — one format per campaign, enforced at creation wizard
  SB_PRODUCT_COLLECTION
  SB_STORE_SPOTLIGHT
  SB_VIDEO

  // SD — targeting tactic chosen at campaign creation
  SD_CONTEXTUAL
  SD_AUDIENCES
  SD_OPTIMIZED
}

enum EntityStatus {
  ENABLED
  PAUSED
  ARCHIVED
  PENDING_REVIEW // SB creative under Amazon's ~72hr moderation
  REJECTED
}

enum BiddingStrategy {
  DYNAMIC_DOWN_ONLY
  DYNAMIC_UP_AND_DOWN
  FIXED_BID
}

enum BudgetType {
  DAILY
  LIFETIME
}

enum MatchType {
  BROAD
  PHRASE
  EXACT
}

enum NegativeMatchType {
  NEGATIVE_EXACT
  NEGATIVE_PHRASE
}

enum TargetingExpressionType {
  // Used for SP manual-product and SD contextual targeting
  ASIN
  CATEGORY
  BRAND
  PRICE_RANGE
  STAR_RATING
  PRIME_SHIPPING
}

enum AudienceSegmentType {
  VIEWS_REMARKETING
  PURCHASES_REMARKETING
  IN_MARKET
  LIFESTYLE
  COMPETITOR_VIEWERS
}

enum PlacementType {
  TOP_OF_SEARCH
  REST_OF_SEARCH
  PRODUCT_PAGES
  HOME_PAGE
  DETAIL_PAGE
}

enum AdEligibilityStatus {
  READY
  NOT_READY_OUT_OF_STOCK
  NOT_READY_NOT_BUY_BOX
  NOT_READY_SUPPRESSED_LISTING
  NOT_READY_POLICY
}

enum RuleConditionMetric {
  ACOS
  SPEND
  CLICKS
  IMPRESSIONS
  ORDERS
  CTR
  CVR
}

enum RuleAction {
  PAUSE
  ENABLE
  ADJUST_BID_PERCENT
  ADJUST_BUDGET_PERCENT
  SEND_ALERT
}

enum ScenarioDifficulty {
  BEGINNER
  INTERMEDIATE
  ADVANCED
}

// ----------------------------------------------------------------------------
// CORE ACCOUNT / TRAINEE LAYER
// ----------------------------------------------------------------------------

model TraineeAccount {
  id            String          @id @default(uuid())
  name          String
  email         String          @unique
  createdAt     DateTime        @default(now())

  // Each trainee gets one or more sandbox seller accounts to practice in
  sellerAccounts MockSellerAccount[]
  scenarioRuns   ScenarioRun[]
}

// A "seller account" is the sandboxed Seller Central / Ads Console instance
// the trainee operates inside. Lets us seed different account states
// (e.g. an account with a Brand Store vs. one without, for SB Store Spotlight
// eligibility scenarios).
model MockSellerAccount {
  id              String   @id @default(uuid())
  trainee         TraineeAccount @relation(fields: [traineeId], references: [id])
  traineeId       String
  marketplaceName String   @default("Amazon.com")
  hasBrandStore   Boolean  @default(false)
  storeSubpageCount Int    @default(0) // gates SB Store Spotlight eligibility
  isBrandRegistered Boolean @default(false) // gates SB + SD eligibility
  createdAt       DateTime @default(now())

  portfolios      Portfolio[]
  campaigns       Campaign[]
  products        MockProduct[]
}

// Mock catalog — minimal ASIN data needed to drive eligibility logic and
// realistic-looking ads (price, stock, buy box).
model MockProduct {
  id               String   @id @default(uuid())
  sellerAccount    MockSellerAccount @relation(fields: [sellerAccountId], references: [id])
  sellerAccountId  String
  asin             String
  title            String
  imageUrl         String?
  price            Decimal  @db.Decimal(10, 2)
  inStock          Boolean  @default(true)
  hasBuyBox        Boolean  @default(true)
  category         String
  brand            String
  starRating       Decimal  @db.Decimal(2, 1) @default(4.0)
  isPrime          Boolean  @default(true)

  ads              Ad[]
  productTargets   ProductTarget[]
}

// ----------------------------------------------------------------------------
// PORTFOLIO
// ----------------------------------------------------------------------------

model Portfolio {
  id              String   @id @default(uuid())
  sellerAccount   MockSellerAccount @relation(fields: [sellerAccountId], references: [id])
  sellerAccountId String
  name            String
  budgetCapAmount Decimal? @db.Decimal(10, 2) // optional portfolio-level cap
  budgetCapStart  DateTime?
  budgetCapEnd    DateTime?
  status          EntityStatus @default(ENABLED)
  createdAt       DateTime @default(now())

  campaigns       Campaign[]
}

// ----------------------------------------------------------------------------
// CAMPAIGN
// ----------------------------------------------------------------------------

model Campaign {
  id              String          @id @default(uuid())
  sellerAccount   MockSellerAccount @relation(fields: [sellerAccountId], references: [id])
  sellerAccountId String
  portfolio       Portfolio?      @relation(fields: [portfolioId], references: [id])
  portfolioId     String?

  name            String
  campaignType    CampaignType
  subtype         CampaignSubtype
  status          EntityStatus    @default(ENABLED)

  budgetType      BudgetType      @default(DAILY)
  budgetAmount    Decimal         @db.Decimal(10, 2)
  isBudgetLimited Boolean         @default(false) // "campaign limited by budget" flag

  startDate       DateTime        @default(now())
  endDate         DateTime?

  biddingStrategy BiddingStrategy @default(DYNAMIC_DOWN_ONLY) // SP-relevant
  bidOptimizationGoal String?     // SD-relevant: "page visits" | "conversions" | "brand awareness"

  // SB-specific landing page choice (only relevant when campaignType = SB)
  sbLandingPageType String?       // "BRAND_STORE" | "CUSTOM_LANDING_PAGE" | "STORE_SUBPAGE" | "PRODUCT_DETAIL"
  sbStoreSubpageIds String[]      // up to 3 for Store Spotlight

  createdAt       DateTime        @default(now())
  updatedAt       DateTime        @updatedAt

  adGroups            AdGroup[]
  placementAdjustments PlacementBidAdjustment[]
  negativeKeywords     NegativeKeyword[]      @relation("CampaignNegativeKeywords")
  negativeProductTargets NegativeProductTarget[] @relation("CampaignNegativeProductTargets")
  performanceMetrics   PerformanceMetric[]
  scenarioChecks       ScenarioCheck[]
}

// Top of Search / Rest of Search / Product Pages bid % adjustments — SP only
model PlacementBidAdjustment {
  id           String        @id @default(uuid())
  campaign     Campaign      @relation(fields: [campaignId], references: [id])
  campaignId   String
  placement    PlacementType
  percentage   Int           // e.g. +150 means 150% bid increase
}

// ----------------------------------------------------------------------------
// AD GROUP
// ----------------------------------------------------------------------------

model AdGroup {
  id              String   @id @default(uuid())
  campaign        Campaign @relation(fields: [campaignId], references: [id])
  campaignId      String
  name            String
  status          EntityStatus @default(ENABLED)
  defaultBid      Decimal  @db.Decimal(10, 2)
  createdAt       DateTime @default(now())

  ads             Ad[]
  keywords        Keyword[]
  productTargets  ProductTarget[]
  audienceTargets AudienceTarget[]
  negativeKeywords NegativeKeyword[] @relation("AdGroupNegativeKeywords")
  negativeProductTargets NegativeProductTarget[] @relation("AdGroupNegativeProductTargets")
  performanceMetrics PerformanceMetric[]
}

// ----------------------------------------------------------------------------
// ADS / CREATIVE
// ----------------------------------------------------------------------------

// Covers SP ad (just an ASIN + optional video), and SB creative fields
// (logo/headline/format-specific assets). One table, nullable SB-only fields
// keeps the schema from fragmenting into per-type creative tables.
model Ad {
  id                String   @id @default(uuid())
  adGroup           AdGroup  @relation(fields: [adGroupId], references: [id])
  adGroupId         String
  product           MockProduct? @relation(fields: [productId], references: [id])
  productId         String?
  status            EntityStatus @default(ENABLED)
  eligibilityStatus AdEligibilityStatus @default(READY)

  // SP video add-on (independent of SB video campaign type)
  hasSpVideo        Boolean  @default(false)
  spVideoUrls       String[] // 1–5 product feature videos

  // SB creative fields — null unless parent campaign.campaignType = SB
  sbHeadline        String?  // max 50 chars, enforced at app layer
  sbLogoUrl         String?
  sbCustomImageUrl  String?
  sbFeaturedProductIds String[] // Product Collection: 3+; Video: exactly 1
  sbReviewStatus    EntityStatus? // PENDING_REVIEW / ENABLED / REJECTED, ~72hr mock delay

  createdAt         DateTime @default(now())

  performanceMetrics PerformanceMetric[]
}

// ----------------------------------------------------------------------------
// TARGETING — SP keywords, SP/SD product targeting, SD audiences
// ----------------------------------------------------------------------------

model Keyword {
  id          String    @id @default(uuid())
  adGroup     AdGroup   @relation(fields: [adGroupId], references: [id])
  adGroupId   String
  text        String
  matchType   MatchType
  bid         Decimal   @db.Decimal(10, 2)
  status      EntityStatus @default(ENABLED)
  // Amazon's suggested bid range, shown next to the bid field in the UI
  suggestedBidLow  Decimal? @db.Decimal(10, 2)
  suggestedBidHigh Decimal? @db.Decimal(10, 2)
  createdAt   DateTime  @default(now())

  performanceMetrics PerformanceMetric[]
  promotedFromSearchTerm SearchTermReportRow? @relation("PromotedKeyword")
}

// SP manual product targeting AND SD contextual product/category targeting
model ProductTarget {
  id            String   @id @default(uuid())
  adGroup       AdGroup  @relation(fields: [adGroupId], references: [id])
  adGroupId     String
  expressionType TargetingExpressionType
  // exactly one of these populated depending on expressionType
  targetProduct MockProduct? @relation(fields: [targetProductId], references: [id])
  targetProductId String?
  categoryName  String?
  brandName     String?
  priceMin      Decimal? @db.Decimal(10, 2)
  priceMax      Decimal? @db.Decimal(10, 2)
  minStarRating Decimal? @db.Decimal(2, 1)
  primeOnly     Boolean?

  bid           Decimal  @db.Decimal(10, 2)
  status        EntityStatus @default(ENABLED)
  createdAt     DateTime @default(now())

  performanceMetrics PerformanceMetric[]
}

// SD audiences targeting only
model AudienceTarget {
  id              String   @id @default(uuid())
  adGroup         AdGroup  @relation(fields: [adGroupId], references: [id])
  adGroupId       String
  segmentType     AudienceSegmentType
  lookbackDays    Int?     // e.g. 14, 30, 90 — relevant for remarketing segments
  excludedProductIds String[] // e.g. exclude purchased ASIN on cross-sell campaigns
  bid             Decimal  @db.Decimal(10, 2)
  status          EntityStatus @default(ENABLED)
  createdAt       DateTime @default(now())

  performanceMetrics PerformanceMetric[]
}

// ----------------------------------------------------------------------------
// NEGATIVE TARGETING — campaign-level OR ad-group-level (both allowed, only
// one parent populated)
// ----------------------------------------------------------------------------

model NegativeKeyword {
  id          String   @id @default(uuid())
  campaign    Campaign? @relation("CampaignNegativeKeywords", fields: [campaignId], references: [id])
  campaignId  String?
  adGroup     AdGroup?  @relation("AdGroupNegativeKeywords", fields: [adGroupId], references: [id])
  adGroupId   String?
  text        String
  matchType   NegativeMatchType
  createdAt   DateTime @default(now())
}

model NegativeProductTarget {
  id          String   @id @default(uuid())
  campaign    Campaign? @relation("CampaignNegativeProductTargets", fields: [campaignId], references: [id])
  campaignId  String?
  adGroup     AdGroup?  @relation("AdGroupNegativeProductTargets", fields: [adGroupId], references: [id])
  adGroupId   String?
  targetAsin  String
  createdAt   DateTime @default(now())
}

// ----------------------------------------------------------------------------
// SEARCH TERM REPORT — feeds the harvesting workflow (promote to keyword /
// add as negative, directly from a report row)
// ----------------------------------------------------------------------------

model SearchTermReportRow {
  id               String   @id @default(uuid())
  adGroup          AdGroup  @relation(fields: [adGroupId], references: [id])
  adGroupId        String
  searchTerm       String
  matchedKeyword   Keyword? @relation(fields: [matchedKeywordId], references: [id])
  matchedKeywordId String?
  date             DateTime @db.Date

  impressions      Int      @default(0)
  clicks           Int      @default(0)
  spend            Decimal  @db.Decimal(10, 2) @default(0)
  orders           Int      @default(0)
  sales            Decimal  @db.Decimal(10, 2) @default(0)

  // set once a trainee acts on this row, for scenario validation
  promotedToKeyword     Keyword? @relation("PromotedKeyword", fields: [promotedKeywordId], references: [id])
  promotedKeywordId     String?  @unique
  addedAsNegative       Boolean  @default(false)
}

// ----------------------------------------------------------------------------
// PERFORMANCE METRICS — single polymorphic daily rollup, reused at every level
// ----------------------------------------------------------------------------

model PerformanceMetric {
  id           String   @id @default(uuid())
  date         DateTime @db.Date

  // Exactly one of the following parent FKs is populated per row.
  campaign     Campaign? @relation(fields: [campaignId], references: [id])
  campaignId   String?
  adGroup      AdGroup?  @relation(fields: [adGroupId], references: [id])
  adGroupId    String?
  ad           Ad?       @relation(fields: [adId], references: [id])
  adId         String?
  keyword      Keyword?  @relation(fields: [keywordId], references: [id])
  keywordId    String?
  productTarget ProductTarget? @relation(fields: [productTargetId], references: [id])
  productTargetId String?
  audienceTarget AudienceTarget? @relation(fields: [audienceTargetId], references: [id])
  audienceTargetId String?
  placement    PlacementType?  // optional placement-level breakdown

  impressions  Int      @default(0)
  clicks       Int      @default(0)
  spend        Decimal  @db.Decimal(10, 2) @default(0)
  orders       Int      @default(0)
  sales        Decimal  @db.Decimal(10, 2) @default(0)

  // SP video specific (per the "Video" report Amazon exposes)
  fiveSecondViews Int?
}

// ----------------------------------------------------------------------------
// AUTOMATION RULES — simple bulk-rule builder
// ----------------------------------------------------------------------------

model AutomationRule {
  id              String   @id @default(uuid())
  sellerAccount   MockSellerAccount @relation(fields: [sellerAccountId], references: [id])
  sellerAccountId String
  name            String
  scopeCampaignIds String[] // empty = applies account-wide

  conditionMetric    RuleConditionMetric
  conditionOperator  String   // ">" | "<" | ">=" | "<="
  conditionValue     Decimal  @db.Decimal(10, 2)
  lookbackDays       Int      @default(14)

  action          RuleAction
  actionValue     Decimal? @db.Decimal(10, 2) // % for bid/budget adjustments

  isActive        Boolean  @default(true)
  lastRunAt       DateTime?
  createdAt       DateTime @default(now())
}

// ----------------------------------------------------------------------------
// TRAINING LAYER — scenario mode + mistake injection + scoring
// ----------------------------------------------------------------------------

model ScenarioTemplate {
  id              String   @id @default(uuid())
  title           String
  description     String
  difficulty      ScenarioDifficulty
  campaignTypeFocus CampaignType?
  // Seed config: what the mock account should look like before the trainee
  // starts (e.g. hasBrandStore=false, to set up an "impossible" Store
  // Spotlight task for troubleshooting practice)
  seedConfig      Json
  // Machine-checkable success conditions, evaluated against final state
  successCriteria Json
  hasInjectedMistake Boolean @default(false)
  mistakeDescription String?

  runs            ScenarioRun[]
}

model ScenarioRun {
  id              String   @id @default(uuid())
  trainee         TraineeAccount   @relation(fields: [traineeId], references: [id])
  traineeId       String
  template        ScenarioTemplate @relation(fields: [templateId], references: [id])
  templateId      String
  startedAt       DateTime @default(now())
  completedAt     DateTime?
  passed          Boolean?
  score           Int?     // 0–100, weighted by ScenarioCheck importance

  checks          ScenarioCheck[]
}

// Individual pass/fail checkpoints within one scenario run, tied to the
// specific campaign/entity the trainee built — this is what "validates
// against the resulting state object, not screenshots" in practice.
model ScenarioCheck {
  id            String      @id @default(uuid())
  run           ScenarioRun @relation(fields: [runId], references: [id])
  runId         String
  campaign      Campaign?   @relation(fields: [campaignId], references: [id])
  campaignId    String?
  checkLabel    String      // e.g. "Daily budget set to $50"
  passed        Boolean     @default(false)
  weight         Int        @default(1)
}
